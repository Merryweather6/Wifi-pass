        // Handle switching between View 1 and View 2

        function switchView(viewNumber) {

            const v1 = document.getElementById('view1');

            const v2 = document.getElementById('view2');

            

            if (viewNumber === 2) {

                v1.classList.replace('opacity-100', 'opacity-0');

                v1.classList.replace('pointer-events-auto', 'pointer-events-none');

                v2.classList.replace('opacity-0', 'opacity-100');

                v2.classList.replace('pointer-events-none', 'pointer-events-auto');

            } else {

                v2.classList.replace('opacity-100', 'opacity-0');

                v2.classList.replace('pointer-events-auto', 'pointer-events-none');

                v1.classList.replace('opacity-0', 'opacity-100');

                v1.classList.replace('pointer-events-none', 'pointer-events-auto');

            }

        }


        // Handle Password Copying & UI Feedback

        async function copyPwd() {

            const password = "SuperSecretPassword123";

            const btn = document.getElementById('copyBtn');

            const txt = document.getElementById('copyText');

            

            // Execute clipboard copy with fallback for maximum compatibility (iOS/Android/iFrames)

            try {

                if (navigator.clipboard && window.isSecureContext) {

                    await navigator.clipboard.writeText(password);

                } else {

                    const textArea = document.createElement("textarea");

                    textArea.value = password;

                    document.body.appendChild(textArea);

                    textArea.select();

                    document.execCommand('copy');

                    document.body.removeChild(textArea);

                }

            } catch (err) {

                console.error("Failed to copy password", err);

            }


            // Visual Success State (iOS Green)

            btn.classList.remove('bg-[#1c1c1e]', 'border-gray-800');

            btn.classList.add('bg-green-500', 'border-green-400');

            

            txt.classList.remove('tracking-[0.3em]', 'text-2xl', 'mt-1');

            txt.classList.add('tracking-normal', 'text-[22px]', 'mt-0', 'flex', 'items-center', 'gap-2');

            txt.innerHTML = `<svg class="w-6 h-6" fill="none" stroke="currentColor" viewBox="0 0 24 24"><path stroke-linecap="round" stroke-linejoin="round" stroke-width="3" d="M5 13l4 4L19 7"></path></svg>Copied!`;


            // Revert back to original state after 2 seconds

            setTimeout(() => {

                btn.classList.add('bg-[#1c1c1e]', 'border-gray-800');

                btn.classList.remove('bg-green-500', 'border-green-400');

                

                txt.classList.add('tracking-[0.3em]', 'text-2xl', 'mt-1');

                txt.classList.remove('tracking-normal', 'text-[22px]', 'mt-0', 'flex', 'items-center', 'gap-2');

                txt.innerText = "••••••••";

            }, 2000);

        }
